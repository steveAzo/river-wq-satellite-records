\documentclass[border=2pt]{standalone}
\usepackage[T1]{fontenc}
\usepackage{mathptmx}
\usepackage{tikz}
\usetikzlibrary{positioning,arrows.meta,fit}
\begin{document}
\begin{tikzpicture}[
  box/.style={draw, rounded corners=2pt, align=center, font=\small, minimum height=1.05cm, inner sep=4pt, text width=2.75cm},
  arr/.style={-{Stealth[length=2mm]}, thick}]
% node
\node[box] (s) {Sensor readings of one UTC day (T, SC, DO, pH)};
\node[box, right=0.45cm of s] (e) {Encoder: fewest values keeping every reading within tolerance};
\node[box, right=0.45cm of e] (f) {Frame: 23-B messages with day, positions, CRC};
\node[box, right=0.45cm of f] (c) {Record-first code: $D_1$ data + $Q-D_1$ parity};
% link
\node[box, below=1.5cm of c, text width=3.1cm] (l) {One-way satellite link: messages lost without notice};
% server
\node[box, left=0.45cm of l] (d) {Decode: majority header, CRC-checked units, erasure decoding};
\node[box, left=0.45cm of d] (r) {Reconstruct and certify spans from received positions only};
\node[box, left=0.45cm of r] (u) {Decisions: certainly exceeded, certainly not, or undetermined};

\draw[arr] (e) -- (f);

\draw[arr] (c) -- node[right, pos=0.62, font=\footnotesize, align=left]{$Q$ messages\\per day} (l);
\draw[arr] (l) -- (d);
\draw[arr] (d) -- (r);
\draw[arr] (r) -- (u);
\node[draw, dashed, rounded corners, inner sep=5pt, fit=(s)(e)(f)(c), label={[font=\footnotesize]above:Sensor node (knows the readings; never learns what arrived)}] {};
\node[draw, dashed, rounded corners, inner sep=5pt, fit=(u)(r)(d), label={[font=\footnotesize]below:Server (knows only the bytes that arrived and the site profile)}] {};
\end{tikzpicture}
